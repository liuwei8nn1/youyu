## 命名速查

### DDD 对象命名

| 层级 | 类型 | 后缀 | 示例 |
|------|------|------|------|
| 基础设施层 | 数据库对象 | DO | UserDO, OrderDO |
| 领域层 | 聚合根 | Aggregate | UserAggregate, OrderAggregate |
| 领域层 | 实体 | 无 | OrderItem, Address |
| 领域层 | 值对象 | 无 | Money, Email |
| 应用层 | 请求对象 | Request | CreateUserRequest |
| 应用层 | 响应对象 | VO / Response | UserVO, OrderResponse |
| 应用层 | DTO | DTO | UserDTO |
| 接口层 | 控制器 | Controller | UserController |
| 仓储 | 接口 | Repository | UserRepository |
| 仓储 | 实现 | RepositoryImpl | UserRepositoryImpl |
| Mapper | MyBatis | Mapper | UserMapper |

### 包命名

```
com.youyu.{module}.{layer}

com.youyu.user.application.service
com.youyu.user.domain.aggregate
com.youyu.user.domain.repository
com.youyu.user.infrastructure.persistence
```

### DDD 聚合根模板

```java
/**
 * {聚合名称}聚合根
 */
public class {AggregateName}Aggregate {
    
    private {AggregateName}Id id;
    private List<{EntityName}> entities;
    private List<DomainEvent> events = new ArrayList<>();
    
    // 工厂方法
    public static {AggregateName}Aggregate create({Params}) {
        {AggregateName}Aggregate aggregate = new {AggregateName}Aggregate();
        aggregate.id = new {AggregateName}Id(IdGenerator.nextId());
        // 初始化逻辑
        return aggregate;
    }
    
    // 业务方法
    public void {businessMethod}({Params}) {
        // 业务规则验证
        if ({condition}) {
            throw new DomainException("{错误消息}");
        }
        
        // 业务逻辑
        // ...
        
        // 发布领域事件
        registerEvent(new {EventName}(this.id));
    }
    
    // 注册领域事件
    private void registerEvent(DomainEvent event) {
        this.events.add(event);
    }
    
    // 获取并清除事件
    public List<DomainEvent> getAndClearEvents() {
        List<DomainEvent> result = new ArrayList<>(this.events);
        this.events.clear();
        return result;
    }
}
```

### Repository 模板

```java
// 接口 - 领域层
public interface {EntityName}Repository {
    {EntityName}Aggregate findById({IdType} id);
    void save({EntityName}Aggregate aggregate);
    void delete({IdType} id);
}

// 实现 - 基础设施层
@Repository
public class {EntityName}RepositoryImpl implements {EntityName}Repository {
    
    @Autowired
    private {EntityName}Mapper mapper;
    
    @Override
    public {EntityName}Aggregate findById({IdType} id) {
        {EntityName}DO entityDO = mapper.selectById(id);
        if (entityDO == null) {
            return null;
        }
        return {EntityName}Converter.toAggregate(entityDO);
    }
    
    @Override
    public void save({EntityName}Aggregate aggregate) {
        {EntityName}DO entityDO = {EntityName}Converter.toDO(aggregate);
        if (entityDO.getId() == null) {
            mapper.insert(entityDO);
        } else {
            mapper.updateById(entityDO);
        }
    }
    
    @Override
    public void delete({IdType} id) {
        mapper.deleteById(id);
    }
}
```

### 全局异常处理模板

```java
@RestControllerAdvice
@Slf4j
public class GlobalExceptionHandler {
    
    @ExceptionHandler(DomainException.class)
    public Result<Void> handleDomainException(DomainException e) {
        log.warn("业务异常: {}", e.getMessage());
        return Result.fail(BizCode.BIZ_ERROR, e.getMessage());
    }
    
    @ExceptionHandler(ApplicationException.class)
    public Result<Void> handleApplicationException(ApplicationException e) {
        log.error("应用异常", e);
        return Result.fail(BizCode.APP_ERROR, e.getMessage());
    }
    
    @ExceptionHandler(MethodArgumentNotValidException.class)
    public Result<Void> handleValidationException(MethodArgumentNotValidException e) {
        String message = e.getBindingResult().getFieldErrors().stream()
            .map(FieldError::getDefaultMessage)
            .findFirst()
            .orElse("参数验证失败");
        log.warn("参数验证失败: {}", message);
        return Result.fail(BizCode.PARAM_ERROR, message);
    }
    
    @ExceptionHandler(Exception.class)
    public Result<Void> handleException(Exception e) {
        log.error("系统异常", e);
        return Result.fail(BizCode.SYSTEM_ERROR, "系统繁忙，请稍后重试");
    }
}
```

### MQ 可靠发送模板

```java
@Service
@Slf4j
public class {EventName}MessageService {
    
    @Autowired
    private RocketMQTemplate rocketMQTemplate;
    
    @Autowired
    private CompensationService compensationService;
    
    public void sendMessage({EventName}Event event) {
        String messageId = UUID.randomUUID().toString();
        
        try {
            // 1. 保存补偿记录
            compensationService.saveRecord(messageId, event);
            
            // 2. 发送消息
            SendResult result = rocketMQTemplate.syncSend(
                "{topic-name}", 
                MessageBuilder.withPayload(event)
                    .setHeader(MessageConst.PROPERTY_KEYS, messageId)
                    .build()
            );
            
            // 3. 检查发送结果
            if (result.getSendStatus() != SendStatus.SEND_OK) {
                throw new InfrastructureException("消息发送失败");
            }
            
            // 4. 标记为已发送
            compensationService.markAsSent(messageId);
            
            log.info("消息发送成功, messageId: {}", messageId);
            
        } catch (Exception e) {
            log.error("发送消息失败, messageId: {}", messageId, e);
            throw e;
        }
    }
}
```

### 分布式锁模板

```java
@Service
public class {BusinessName}Service {
    
    @Autowired
    private RedissonClient redisson;
    
    public {ReturnType} {businessMethod}({Params}) {
        String lockKey = "{lock-prefix}:" + {key};
        RLock lock = redisson.getLock(lockKey);
        
        try {
            // 尝试获取锁
            boolean locked = lock.tryLock(3, 10, TimeUnit.SECONDS);
            if (!locked) {
                throw new BusinessException("系统繁忙，请稍后重试");
            }
            
            // 执行业务逻辑
            return doBusinessLogic({params});
            
        } catch (InterruptedException e) {
            Thread.currentThread().interrupt();
            throw new BusinessException("系统繁忙");
        } finally {
            // 确保释放锁
            if (lock.isHeldByCurrentThread()) {
                lock.unlock();
            }
        }
    }
}
```
